class Pair{
    int l;
    int r;
    int u;
    int d;
    Pair(int l,int r,int u,int d)
    {
        this.l=l;
        this.r=r;
        this.u=u;
        this.d=d;
    }
}
class Solution {
    public int distinctPoints(String s, int k) {
        HashSet<String>set=new HashSet<>();
        Pair []prefix=new Pair[s.length()];
        for (int i = 0; i <s.length(); i++) {
    char ch = s.charAt(i);
    int l = 0, r = 0, u = 0, d = 0;

    if (i > 0) {
        l = prefix[i-1].l;
        r = prefix[i-1].r;
        u = prefix[i-1].u;
        d = prefix[i-1].d;
    }

    // Increment the count for the current character
    if (ch == 'L') l++;
    if (ch == 'R') r++;
    if (ch == 'U') u++;
    if (ch == 'D') d++;

    prefix[i] = new Pair(l, r, u, d);
}
       if(s.length()==k)
       {
           return 1;
       }
        for(int i=0;i<=s.length()-k;i++)
            {
                int xx[]=new int[2];
                // String str=s.substring(i,k);
                // String str = s.substring(0, i) + s.substring(i + k,s.length());
                 Pair pair1=(i-1>=0)?prefix[i-1]:new Pair(0,0,0,0);
                Pair pair2=prefix[i+k-1];
                Pair pair3=prefix[s.length()-1];
                // for(int j=0;j<str.length();j++)
                //     {
                //         char ch=str.charAt(j);
                        // if(pair1.=='L')
                        // {
                        //     xx[0]=xx[0]-1;
                        // }else if(ch=='U')
                        // {
                        //     xx[1]=xx[1]+1;
                        // }
                        // else if(ch=='D')
                        // {
                        //     xx[1]=xx[1]-1;
                        // }
                        // else
                        // {
                        //     xx[0]=xx[0]+1;
                        // }
                       xx[0]=xx[0]-(pair1.l+(pair3.l-pair2.l));
                     xx[1]=xx[1]+(pair1.u+(pair3.l-pair2.u));
                xx[1]=xx[1]-(pair1.d+(pair3.d-pair2.d));
                xx[0]=xx[0]+(pair1.r+(pair3.r-pair2.r));
                        String ss=xx[0]+"-"+xx[1];
                        set.add(ss);
                    
                    // }
            }
        return set.size();
    }
